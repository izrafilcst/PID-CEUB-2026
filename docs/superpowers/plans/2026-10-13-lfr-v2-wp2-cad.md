# LFR v2 — WP2 Mecânica (build123d) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Modelar em build123d a base de 2 rodas do LFR — base em PETG, abraçadeiras do N20,
berço do suporte 2×18650, barra do QTR-8A com patins, rodas impressas com O-ring, suportes da
eletrônica e cupom de tolerância —, verificada por testes (interferência, envelope, altura do
sensor, vão livre, CG, imprimibilidade) e exportada em STL e STEP.

**Architecture:** `hardware/cad/params.py` é a fonte única de cotas: lê o `specs.py`, aplica o
`medidas.toml` (medidas tiradas das peças em mãos e offsets do cupom) e define as tolerâncias
de PETG. Cada peça é um módulo em `parts/` que constrói o sólido **na posição de montagem**
(origem no centro do eixo motriz, no chão; X para a frente, Y para a esquerda, Z para cima) e
declara sua orientação de impressão. `assembly.py` posiciona peças e envelopes dos
componentes e faz as verificações; `build.py` exporta tudo para `out/`.

**Tech Stack:** Python 3.12, build123d (OCP/OpenCascade) num venv local, `unittest`.

**Spec:** `docs/superpowers/specs/2026-10-13-lfr-v2-retomada-design.md` (§2 e §5).
Plano mestre: `docs/superpowers/plans/2026-10-13-lfr-v2-00-mestre.md`.

  só como ponte ≤ 10 mm.

```python
"""Fonte única de cotas e tolerâncias do CAD (spec §5.2–5.4).

Prioridade: medidas.toml (medido na peça em mãos ou no cupom) > specs.py (fichas) >
valores típicos deste arquivo. Cotas em mm, massas em g.
"""
from __future__ import annotations

import math
import sys
import tomllib
from dataclasses import dataclass, fields, replace
from functools import lru_cache
from pathlib import Path

CAD_DIR = Path(__file__).resolve().parent
REPO = CAD_DIR.parent.parent
TOML_PADRAO = CAD_DIR / "medidas.toml"
sys.path.insert(0, str(REPO / "hardware" / "components"))
sys.path.insert(0, str(REPO / "tools"))

from specs import COMPONENTES  # noqa: E402


def _spec(comp: str, medida: str) -> float:
    return COMPONENTES[comp].medidas[medida].valor


@dataclass(frozen=True)
class Tolerancias:
    deslizante: float = 0.30          # por lado
    pcb: float = 0.20                 # por lado
    furo_m2: float = 2.4
    furo_m3: float = 3.4
    autoatarraxante_m2: float = 1.7
    autoatarraxante_prof: float = 6.0
    inserto_m3_d: float = 4.0
    inserto_m3_prof: float = 6.5
    eixo_d: float = 3.05
    eixo_achatado: float = 2.55
    n20_folga: float = 0.10           # por lado
    comp_furo: float = 0.10           # acréscimo no diâmetro de furos < ⌀6


@dataclass(frozen=True)
class Params:
    tol: Tolerancias
    # ── N20 (ficha 04) ─────────────────────────────────────────────────────
    n20_largura: float                # horizontal (X)
    n20_altura: float                 # vertical (Z) — Pololu: seção 10 × 12 mm

    n20_total_l: float                # face da caixa → fundo do encoder
    n20_conector_l: float             # conector JST-SH além do encoder
    n20_eixo_l: float
    n20_eixo_d: float
    n20_achatado: float
    n20_massa: float
    n20_rpm_6v: float = 600.0
    n20_torque_stall_kgcm: float = 0.5    # 6 V, típico de N20 50:1 genérico
    # ── Bateria ────────────────────────────────────────────────────────────
    celula_d: float = 18.4
    celula_l: float = 65.2
    celula_massa: float = 46.0
    suporte_l: float = 77.0           # ao longo das células (Y no carro)
    suporte_w: float = 41.0           # X no carro
    suporte_h: float = 20.5
    suporte_massa: float = 12.0
    # ── QTR-8A (ficha 03) ──────────────────────────────────────────────────
    qtr_l: float = 0.0
    qtr_w: float = 0.0
    qtr_t: float = 0.0                # PCB + sensores
    qtr_massa: float = 0.0
    qtr_furos_dist: float = 69.85     # entre os furos de fixação nas pontas
    qtr_furo_d: float = 2.2
    # ── Eletrônica ─────────────────────────────────────────────────────────
    perf_l: float = 0.0               # 70 mm, ao longo de X no carro
    perf_w: float = 0.0               # 50 mm, ao longo de Y

    fios_parafusos_massa: float = 10.0

    interruptor_h: float = 3.6

    oring_id: float = 34.0
    oring_cs: float = 3.0
    oring_estiramento: float = 0.03
    oring_massa: float = 0.5
    roda_aro_w: float = 8.5

    lookahead: float = 85.0
    sensor_altura: float = 3.0
    vao_minimo: float = 4.0
    base_z: float = 5.0               # face inferior da base

    base_meia_largura: float = 39.5

    folga_roda_motor: float = 0.6       # > 0,5: folga mínima de peça girando
    parede: float = 2.4
    parede_batente: float = 1.2
    suporte_x_traseira: float = 15.0
    perf_x_frente: float = -10.0
    perf_z: float = 32.0              # face inferior do perfboard
    barra_t: float = 8.0              # espessura da barra: inserto 6,5 + 1,5 por cima
    barra_lingua_x: float = 60.0      # início da língua que apoia na base
    vref: float = 7.4                 # V — tensão de referência do firmware (fábrica)
    # ── Material e máquina ─────────────────────────────────────────────────
    petg_densidade: float = 1.27e-3   # g/mm³
    preenchimento: float = 0.6
    mesa: float = 200.0

    @property
    def roda_cubo_l(self) -> float:
        return self.n20_eixo_l - self.folga_roda_motor

    @property
    def roda_od(self) -> float:

        return fundo + 2 * self.oring_cs * (1 - self.oring_estiramento / 2)

    @property
    def base_topo(self) -> float:

# Cotas △ MEDIR (ou típicas de peça genérica) que o montador deve conferir no medidas.toml
MEDIR_TOML = (
    "suporte_l", "suporte_w", "suporte_h", "celula_d", "n20_altura", "n20_conector_l",
    "qtr_furos_dist", "perf_furo_canto", "interruptor_l", "interruptor_h", "oring_id",
    "oring_cs", "fios_parafusos_massa",
)


def altura_eixo(p: Params) -> float:
    return p.roda_od / 2


def _base() -> Params:
    massa_eletronica = sum(COMPONENTES[c].massa_g
                           for c in ("esp32", "mcp3008", "tb6612", "mp1584", "perfboard"))
    return Params(
        tol=Tolerancias(),
        n20_largura=_spec("n20", "largura_caixa"),
        n20_altura=10.0,
        n20_caixa_l=_spec("n20", "comprimento_caixa_reducao"),
        n20_total_l=_spec("n20", "comprimento_total"),
        n20_conector_l=5.0,
        n20_eixo_l=_spec("n20", "comprimento_eixo"),
        n20_eixo_d=_spec("n20", "diametro_eixo"),
        n20_achatado=_spec("n20", "achatado_eixo"),
        n20_massa=COMPONENTES["n20"].massa_g,
        qtr_l=_spec("qtr8a", "comprimento"),
        qtr_w=_spec("qtr8a", "largura"),
        qtr_t=_spec("qtr8a", "espessura"),
        qtr_massa=COMPONENTES["qtr8a"].massa_g,
        perf_l=_spec("perfboard", "comprimento"),
        perf_w=_spec("perfboard", "largura"),
        perf_t=_spec("perfboard", "espessura"),
        eletronica_massa=massa_eletronica,
    )


def _aplicar(p: Params, secao: dict, alvo: str) -> Params:
    nomes = {f.name for f in fields(Tolerancias if alvo == "tolerancias" else Params)}
    for chave, valor in secao.items():
        if chave not in nomes or chave == "tol":
            raise KeyError(f"medidas.toml [{alvo}]: chave desconhecida '{chave}'")
        if not isinstance(valor, (int, float)) or not math.isfinite(valor) or valor <= 0:
            raise ValueError(f"medidas.toml [{alvo}]: '{chave}' = {valor!r} inválido")
    if alvo == "tolerancias":
        return replace(p, tol=replace(p.tol, **{k: float(v) for k, v in secao.items()}))
    return replace(p, **{k: float(v) for k, v in secao.items()})


def _carregar(caminho: Path) -> Params:
    p = _base()
    if caminho.exists():
        dados = tomllib.loads(caminho.read_text())
        p = _aplicar(p, dados.get("medidas", {}), "medidas")
        p = _aplicar(p, dados.get("tolerancias", {}), "tolerancias")
    return p


@lru_cache(maxsize=None)
def _carregar_padrao() -> Params:
    return _carregar(TOML_PADRAO)


def carregar(caminho_toml: Path | None = None) -> Params:
    """Params com overrides. Sem argumento: medidas.toml padrão (em cache)."""
    if caminho_toml is None:
        return _carregar_padrao()
    return _carregar(Path(caminho_toml))
```

```python
"""Primitivas posicionadas no sistema da montagem (mm; X frente, Y esquerda, Z cima)."""
from __future__ import annotations

from build123d import Align, Box, Cylinder, Part, Pos, Rot

MIN_Z = (Align.CENTER, Align.CENTER, Align.MIN)


def caixa(dx: float, dy: float, dz: float, x: float = 0.0, y: float = 0.0,
          z: float = 0.0) -> Part:
    """Caixa centrada em (x, y), com a face de baixo em z."""
    return Pos(x, y, z) * Box(dx, dy, dz, align=MIN_Z)


def caixa_lim(x0: float, x1: float, y0: float, y1: float, z0: float, z1: float) -> Part:
    """Caixa pelos limites."""
    return caixa(x1 - x0, y1 - y0, z1 - z0, (x0 + x1) / 2, (y0 + y1) / 2, z0)


def cil_z(d: float, h: float, x: float = 0.0, y: float = 0.0, z: float = 0.0) -> Part:
    """Cilindro vertical com a base em z."""
    return Pos(x, y, z) * Cylinder(d / 2, h, align=MIN_Z)


def cil_y(d: float, comp: float, x: float = 0.0, y0: float = 0.0, z: float = 0.0) -> Part:
    """Cilindro com eixo ao longo de +Y, de y0 a y0 + comp."""
    return Pos(x, y0, z) * Rot(-90, 0, 0) * Cylinder(d / 2, comp, align=MIN_Z)


def na_mesa(part: Part, rot: Rot) -> Part:
    """Gira para a orientação de impressão e apoia na mesa (z mín = 0), centrado em XY."""
    girada = rot * part
    bb = girada.bounding_box()
    return Pos(-(bb.min.X + bb.max.X) / 2, -(bb.min.Y + bb.max.Y) / 2, -bb.min.Z) * girada
```

```python
"""Roda impressa em PETG com canal para O-ring (spec §5.4).

Referencial da peça: eixo de rotação = Z, face externa em z = 0 (vai na mesa), cubo
crescendo para +Z (lado do motor). O furo em D tem o achatado voltado para +Y.
"""
from __future__ import annotations

import math
from dataclasses import dataclass
from functools import lru_cache

from build123d import Align, Axis, Box, Circle, Cylinder, Part, Plane, Polygon, Pos, Rot, revolve

from params import Params, carregar
from parts.common import MIN_Z

ROT_IMPRESSAO = Rot(0, 0, 0)          # face externa na mesa
FAIXA_OD = (38.0, 44.0)               # spec §5.4
ORINGS_METRICOS = [(30, 3), (31, 3), (32, 3), (33, 3), (34, 3), (35, 3), (36, 3),
                   (32, 2.5), (33, 2.5), (34, 2.5), (35, 2.5), (36, 2.5)]
CARGA = 0.8                            # rotação sob carga / sem carga
TAN_TETO = math.tan(math.radians(55))  # teto do canal a 55° da horizontal (35° de balanço)
Z_CANAL = 1.2                          # flanco de baixo do canal, acima da face externa


@dataclass(frozen=True)
class GeometriaORing:
    id: float
    cs: float
    fundo_d: float
    cs_estirada: float
    diametro_externo: float
    profundidade: float
    largura: float
    aro_d: float


def geometria_oring(id_mm: float, cs_mm: float, estiramento: float) -> GeometriaORing:
    fundo = id_mm * (1 + estiramento)
    cs_e = cs_mm * (1 - estiramento / 2)
    prof = 0.72 * cs_mm                       # 70–75 % da seção (spec §5.4)
    return GeometriaORing(id_mm, cs_mm, fundo, cs_e, fundo + 2 * cs_e, prof, 1.25 * cs_mm,
                          fundo + 2 * prof)


def escolher_oring(alvo_od: float = 41.0, estiramento: float = 0.03,
                   faixa: tuple[float, float] = FAIXA_OD) -> GeometriaORing:
    candidatos = [geometria_oring(i, c, estiramento) for i, c in ORINGS_METRICOS]
    validos = [g for g in candidatos if faixa[0] <= g.diametro_externo <= faixa[1]]
    return min(validos, key=lambda g: abs(g.diametro_externo - alvo_od))


def verificar_oring(p: Params) -> GeometriaORing:
    g = geometria_oring(p.oring_id, p.oring_cs, p.oring_estiramento)
    if not FAIXA_OD[0] <= g.diametro_externo <= FAIXA_OD[1]:
        raise ValueError(f"O-ring {p.oring_id} × {p.oring_cs} dá roda de "
                         f"{g.diametro_externo:.1f} mm, fora de {FAIXA_OD[0]:.0f}–"
                         f"{FAIXA_OD[1]:.0f} mm")
    return g


def velocidade_m_s(od_mm: float, rpm: float) -> float:
    return math.pi * od_mm / 1000.0 * rpm / 60.0


def rpm_sob_carga(p: Params) -> float:
    """Rotação de saída na tensão de referência do firmware (vref), com carga típica."""
    return p.n20_rpm_6v * (p.vref / 6.0) * CARGA


def aceleracao_m_s2(p: Params, massa_g: float, fracao_velocidade: float = 0.5) -> float:
    """Aceleração das duas rodas a uma fração da rotação livre (curva torque × rotação linear)."""
    torque_nm = p.n20_torque_stall_kgcm * 0.0980665 * (p.vref / 6.0)
    forca = 2 * torque_nm * (1 - fracao_velocidade) / (p.roda_od / 2000.0)
    return forca / (massa_g / 1000.0)


@lru_cache(maxsize=None)
def fazer_roda(p: Params | None = None) -> Part:
    p = p or carregar()
    g = verificar_oring(p)
    r_aro = g.aro_d / 2
    parede_aro = g.profundidade + 1.2                    # 1,2 mm de PETG sob o canal
    aro = Cylinder(r_aro, p.roda_aro_w, align=MIN_Z) - \
        Cylinder(r_aro - parede_aro, p.roda_aro_w, align=MIN_Z)
    disco = Cylinder(r_aro - parede_aro + 0.5, 2.0, align=MIN_Z)
    cubo = Cylinder(p.roda_cubo_d / 2, p.roda_cubo_l, align=MIN_Z)
    roda = aro + disco + cubo

    # Canal do O-ring: flanco de baixo horizontal, teto a 55° → imprime sem suporte
    z0 = Z_CANAL
    zt = z0 + g.largura - g.profundidade / 2
    r_in = r_aro - g.profundidade
    perfil = Plane.XZ * Polygon((r_in, z0), (r_aro + 1, z0),
                                (r_aro + 1, zt + (g.profundidade + 1) * TAN_TETO), (r_in, zt),
                                align=None)
    roda -= revolve(perfil, Axis.Z)

    # Furo em D passante (achatado voltado para +Y)
    r = p.tol.eixo_d / 2
    manter = Pos(0, -r, -1) * Box(2 * r + 1, p.tol.eixo_achatado, p.roda_cubo_l + 4,
                                  align=(Align.CENTER, Align.MIN, Align.MIN))
    furo_d = Pos(0, 0, -1) * Cylinder(r, p.roda_cubo_l + 2, align=MIN_Z) & manter
    roda -= furo_d

    # Furo radial para M2 autoatarraxante (trava opcional contra o achatado)
    trava = Pos(0, 0, p.roda_cubo_l - 3.0) * Rot(-90, 0, 0) * \
        Cylinder(p.tol.autoatarraxante_m2 / 2, p.roda_cubo_d, align=MIN_Z)
    return roda - trava


@lru_cache(maxsize=None)
def fazer_oring(p: Params | None = None) -> Part:
    """O-ring montado (toro), no mesmo referencial da roda."""
    p = p or carregar()
    g = verificar_oring(p)
    perfil = Plane.XZ * Pos(g.fundo_d / 2 + g.cs_estirada / 2, p.roda_aro_w / 2) * \
        Circle(g.cs_estirada / 2)
    return revolve(perfil, Axis.Z)


def paredes(p: Params | None = None) -> dict[str, float]:
    p = p or carregar()
    g = verificar_oring(p)
    topo_canal = Z_CANAL + g.largura - g.profundidade / 2 + g.profundidade * TAN_TETO
    return {
        "aro_sob_canal": 1.2,
        "disco": 2.0,
        "cubo_ao_furo": (p.roda_cubo_d - p.tol.eixo_d) / 2,
        "ressalto_aro": p.roda_aro_w - topo_canal,
        "flanco_externo": Z_CANAL,
    }
```

O toro do O-ring fica centrado na meia largura do aro só para fins de verificação de folga e
contato com o chão; o canal real é assimétrico (teto a 55°). O disco (z 0–2) termina 0,7 mm

```python
"""Envelopes das peças compradas, na posição de montagem (spec §5.2, §5.4)."""
from __future__ import annotations

from build123d import Part

from params import Params, altura_eixo
from parts.common import caixa, caixa_lim, cil_z

CALCO_T = 0.5  # mm por calço sob o patim


def motor(p: Params, lado: int) -> Part:
    """Corpo do N20 + conector traseiro. lado = +1 (esquerdo, y > 0) ou -1 (direito)."""
    zc = altura_eixo(p)
    y_face = lado * p.motor_face_y
    y_fundo = y_face - lado * p.n20_total_l
    y_conector = y_fundo - lado * p.n20_conector_l
    corpo = caixa_lim(-p.n20_largura / 2, p.n20_largura / 2, min(y_face, y_fundo),
                      max(y_face, y_fundo), zc - p.n20_altura / 2, zc + p.n20_altura / 2)
    conector = caixa_lim(-4.8, 4.8, min(y_fundo, y_conector), max(y_fundo, y_conector),
                         zc - 2.0, zc + 2.0)
    return corpo + conector


def suporte_bateria(p: Params) -> Part:
    x0 = p.suporte_x_traseira
    return caixa_lim(x0, x0 + p.suporte_w, -p.suporte_l / 2, p.suporte_l / 2,
                     p.base_topo, p.base_topo + p.suporte_h)


def qtr(p: Params, calcos: int = 0) -> Part:
    z0 = p.sensor_altura + calcos * CALCO_T
    return caixa(p.qtr_w, p.qtr_l, p.qtr_t, x=p.lookahead, z=z0)


def perfboard(p: Params) -> Part:
    """Perfboard + volume dos componentes por cima + 2 mm de terminais por baixo."""
    x1 = p.perf_x_frente
    x0 = x1 - p.perf_l
    placa = caixa_lim(x0, x1, -p.perf_w / 2, p.perf_w / 2, p.perf_z, p.perf_z + p.perf_t)
    cima = caixa_lim(x0 + 2, x1 - 2, -p.perf_w / 2 + 2, p.perf_w / 2 - 2,
                     p.perf_z + p.perf_t, p.perf_z + p.perf_t + p.eletronica_h)
    baixo = caixa_lim(x0 + 8, x1 - 8, -p.perf_w / 2 + 8, p.perf_w / 2 - 8,
                      p.perf_z - 2.0, p.perf_z)   # terminais longe das colunas de fixação
    return placa + cima + baixo


def massas(p: Params) -> list[tuple[str, float, Part]]:
    """Peças compradas com massa e sólido de referência (centro de massa = centróide)."""
    zc = altura_eixo(p)
    return [
        ("N20 esquerdo", p.n20_massa, motor(p, +1)),
        ("N20 direito", p.n20_massa, motor(p, -1)),
        ("2 células 18650", 2 * p.celula_massa, suporte_bateria(p)),
        ("Suporte 2×18650", p.suporte_massa, suporte_bateria(p)),
        ("QTR-8A", p.qtr_massa, qtr(p)),
        ("Eletrônica no perfboard", p.eletronica_massa, perfboard(p)),
        ("Fios e parafusos", p.fios_parafusos_massa, perfboard(p)),
        ("O-rings", 2 * p.oring_massa, cil_z(1, 1, 0, 0, zc)),
    ]
```

```python
"""Abraçadeira do N20 sobre a caixa de redução (spec §5.4).

Prende a metade de cima da caixa; a de baixo fica no berço da base. Dois M2 autoatarraxantes
nas orelhas. Entre a orelha e o berço sobra APERTO mm: o parafuso puxa a abraçadeira e prensa
o motor. Imprime de lado (perfil em U de pé): sem ponte nem balanço.
"""
from __future__ import annotations

from build123d import Part, Rot, mirror, Plane

from params import Params, altura_eixo
from parts.common import caixa_lim, cil_z

ROT_IMPRESSAO = Rot(90, 0, 0)   # eixo Y da montagem vira o Z da impressora
APERTO = 0.3                    # mm de folga orelha–berço antes de apertar
ORELHA_L = 5.0                  # largura de cada orelha em X
ORELHA_T = 3.0                  # espessura da orelha


def _meia_largura_canal(p: Params) -> float:
    return p.n20_largura / 2 + p.tol.n20_folga


def orelha_x(p: Params) -> float:
    """Centro do furo da orelha, |x|."""
    return _meia_largura_canal(p) + p.parede + ORELHA_L / 2


def furos_parafuso(p: Params, lado: int) -> list[tuple[float, float]]:
    y = lado * (p.motor_face_y - p.n20_caixa_l / 2)
    return [(-orelha_x(p), y), (orelha_x(p), y)]


def fazer(p: Params, lado: int) -> Part:
    zc = altura_eixo(p)
    w = _meia_largura_canal(p)
    topo_canal = zc + p.n20_altura / 2 + p.tol.n20_folga
    topo = topo_canal + p.parede
    y1 = p.motor_face_y - APERTO     # não encosta na parede de batente da base
    y0 = p.motor_face_y - p.n20_caixa_l
    corpo = caixa_lim(-w - p.parede, w + p.parede, y0, y1, zc, topo)
    canal = caixa_lim(-w, w, y0 - 1, y1 + 1, zc - 1, topo_canal)
    ox = orelha_x(p)
    orelhas = (caixa_lim(-ox - ORELHA_L / 2, -w - p.parede, y0, y1, zc, zc + ORELHA_T) +
               caixa_lim(w + p.parede, ox + ORELHA_L / 2, y0, y1, zc, zc + ORELHA_T))
    peca = corpo + orelhas - canal
    for x, y in furos_parafuso(p, +1):
        peca -= cil_z(p.tol.furo_m2, ORELHA_T + 2, x, y, zc - 1)
    return peca if lado > 0 else mirror(peca, about=Plane.XZ)


def paredes(p: Params) -> dict[str, float]:
    return {"lateral": p.parede, "topo": p.parede, "orelha": ORELHA_T,
            "orelha_ao_furo": (ORELHA_L - p.tol.furo_m2) / 2}
```

```python
"""Barra do QTR-8A (spec §5.4–5.5).

Placa em T com espessura barra_t: a viga transversal segura o QTR por dois postes M2 e os
patins nas pontas; a língua vai para trás e apoia na base, presa por 2 M3 que sobem pela base
até insertos térmicos na língua. Imprime com a face de cima na mesa.
"""
from __future__ import annotations

from build123d import Part, Rot

from params import Params
from parts.common import caixa_lim, cil_z
from parts.components import CALCO_T

ROT_IMPRESSAO = Rot(180, 0, 0)
VIGA_X = 17.5          # largura da viga em X
PATIM_Y = 6.0          # largura do patim em Y
PATIM_X = 10.0
POSTE_D = 5.0
LINGUA_Y = 30.0


def barra_z0(p: Params, calcos: int = 0) -> float:
    """Face de baixo da barra: QTR (sensor_altura + qtr_t) + postes de 1,8 mm."""
    return p.sensor_altura + p.qtr_t + 1.82 + calcos * CALCO_T


def y_patim(p: Params) -> float:
    return p.qtr_l / 2 + 2.0 + PATIM_Y / 2 + 3.0


def furos_base(p: Params) -> list[tuple[float, float]]:
    x = p.barra_lingua_x + 6.0
    return [(x, -8.0), (x, 8.0)]


def fazer(p: Params, calcos: int = 0) -> Part:
    z0 = barra_z0(p, calcos)
    z1 = z0 + p.barra_t
    xa = p.lookahead
    ya = y_patim(p) + PATIM_Y / 2
    viga = caixa_lim(xa - VIGA_X / 2, xa + VIGA_X / 2, -ya, ya, z0, z1)
    lingua = caixa_lim(p.barra_lingua_x, xa, -LINGUA_Y / 2, LINGUA_Y / 2,
                       max(z0, p.base_topo), z1)
    barra = viga + lingua
    # Postes do QTR (descem até a face de cima do QTR) com furo para M2 autoatarraxante
    z_qtr_topo = p.sensor_altura + calcos * CALCO_T + p.qtr_t
    for y in (-p.qtr_furos_dist / 2, p.qtr_furos_dist / 2):
        barra += cil_z(POSTE_D, z0 - z_qtr_topo, xa, y, z_qtr_topo)
        barra -= cil_z(p.tol.autoatarraxante_m2, p.tol.autoatarraxante_prof, xa, y, z_qtr_topo)
    # Furos dos patins (M2 de cima para baixo) e insertos M3 da língua (de baixo)
    for y in (-y_patim(p), y_patim(p)):
        barra -= cil_z(p.tol.furo_m2, p.barra_t + 2, xa, y, z0 - 1)
    for x, y in furos_base(p):
        barra -= cil_z(p.tol.inserto_m3_d, p.tol.inserto_m3_prof, x, y, max(z0, p.base_topo))
    return barra


def paredes(p: Params) -> dict[str, float]:
    return {"barra": p.barra_t, "poste_ao_furo": (POSTE_D - p.tol.autoatarraxante_m2) / 2,
            "lingua_sobre_inserto": p.barra_t - p.tol.inserto_m3_prof}
```

O teste `test_qtr_encosta_nos_postes_sem_invadir` exige que os postes toquem o topo do QTR
(distância zero) sem volume de interseção. Com `barra_t = 8` sobra 1,5 mm de PETG sobre o
inserto M3 da língua.

**Calços e nivelamento.** A barra é parafusada na base; se o calço fosse só sob os patins, a
frente subiria e o chassi inclinaria. Por isso cada passo de ajuste usa **um calço sob cada
patim e um sob a língua** (`calco_lingua`): a barra sobe 0,5 mm em relação à base e aos patins
ao mesmo tempo, e o chassi continua nivelado. Com `calcos = n`, `barra_z0` sobe 0,5·n e a
língua deixa um vão de 0,5·n sobre a base, ocupado pelos calços.

```python
def calco_lingua(p: Params, n: int = 1) -> Part:
    """n calços de 0,5 mm entre a língua e a base, com os dois furos M3 de passagem."""
    x_fim = furos_base(p)[0][0] + 6.0
    c = caixa_lim(p.barra_lingua_x, x_fim, -LINGUA_Y / 2, LINGUA_Y / 2, 0.0, 0.5 * n)
    for x, y in furos_base(p):
        c -= cil_z(p.tol.furo_m3, 2, x, y, -1)
    return c
```

```python
"""Patim substituível (spec §5.5): calota de raio RAIO_CALOTA no contato, fixo por um M2 que
desce pela barra. A calota faz o papel da rampa de entrada (≈ 30° num degrau de 0,5 mm)."""
from __future__ import annotations

from build123d import Part, Pos, Rot, Sphere

from params import Params
from parts.common import caixa, caixa_lim, cil_z
from parts.sensor_bar import PATIM_X, PATIM_Y, barra_z0, y_patim

ROT_IMPRESSAO = Rot(180, 0, 0)   # face de fixação na mesa, calota para cima
RAIO_CALOTA = 4.0


def fazer(p: Params, lado: int, calcos: int = 0) -> Part:
    h = barra_z0(p, calcos) - calcos * 0.5        # o calço ocupa o resto da altura
    x = p.lookahead

    r = RAIO_CALOTA
    corpo = caixa_lim(x - PATIM_X / 2, x + PATIM_X / 2, y - PATIM_Y / 2, y + PATIM_Y / 2, r, h)
    calota = Pos(x, y, r) * Sphere(r) & caixa_lim(x - r, x + r, y - PATIM_Y / 2,
                                                  y + PATIM_Y / 2, 0, r)

    return patim - cil_z(p.tol.autoatarraxante_m2, p.tol.autoatarraxante_prof, x, y,
                         h - p.tol.autoatarraxante_prof)


def calco(p: Params) -> Part:

def paredes(p: Params) -> dict[str, float]:
    return {"patim_ao_furo": (PATIM_Y - p.tol.autoatarraxante_m2) / 2, "calco": 0.5}
```

```python
"""Berço do suporte 2×18650 (spec §5.4). Feição unida à base, impressa junto com ela."""
from __future__ import annotations

from build123d import Part

from params import Params
from parts.common import caixa_lim

PAREDE_H = 5.0
COLUNA_X = 6.0
ABA = 1.5
ABA_T = 1.2
PONTA = 6.0      # espessura útil da parede de ponta do suporte (sem célula embaixo)


def _limites(p: Params):
    f = p.tol.deslizante
    x0 = p.suporte_x_traseira - f
    x1 = p.suporte_x_traseira + p.suporte_w + f
    y1 = p.suporte_l / 2 + f
    return x0, x1, y1


def feicao(p: Params) -> Part:
    """Piso (estende a placa da base sob o berço) + paredes e abas."""
    x0, x1, y1 = _limites(p)
    t = p.parede
    piso = caixa_lim(x0 - t, x1 + t, -y1 - t, y1 + t, p.base_z, p.base_topo)
    return piso + paredes_e_abas(p)


def paredes_e_abas(p: Params) -> Part:
    x0, x1, y1 = _limites(p)
    z0 = p.base_topo
    t = p.parede
    fora = caixa_lim(x0 - t, x1 + t, -y1 - t, y1 + t, z0, z0 + PAREDE_H)
    dentro = caixa_lim(x0, x1, -y1, y1, z0 - 1, z0 + PAREDE_H + 1)
    berco = fora - dentro
    z_aba = z0 + p.suporte_h + p.tol.deslizante
    for xc in (x0 + 4 + COLUNA_X / 2, x1 - 4 - COLUNA_X / 2):
        for s in (+1, -1):
            yp = s * (y1 + t / 2)
            coluna = caixa_lim(xc - COLUNA_X / 2, xc + COLUNA_X / 2,
                               min(yp - t / 2, yp + t / 2), max(yp - t / 2, yp + t / 2),
                               z0, z_aba + ABA_T)
            y_aba0 = s * y1
            y_aba1 = s * (y1 - ABA)
            aba = caixa_lim(xc - COLUNA_X / 2, xc + COLUNA_X / 2, min(y_aba0, y_aba1),
                            max(y_aba0, y_aba1), z_aba, z_aba + ABA_T)
            berco += coluna + aba
    return berco


def area_livre_topo(p: Params) -> Part:
    """Prisma acima das células (entre as paredes de ponta): tem que ficar vazio."""
    x0, x1, y1 = _limites(p)
    z = p.base_topo + p.suporte_h
    return caixa_lim(x0, x1, -(y1 - PONTA), y1 - PONTA, z, z + 40.0)


def paredes(p: Params) -> dict[str, float]:
    return {"parede": p.parede, "aba": ABA_T, "coluna": p.parede}
```

```python
from build123d import Part

from params import Params
from parts.common import caixa_lim, cil_z

def furos_perfboard(p: Params) -> list[tuple[float, float]]:
    x1 = p.perf_x_frente - p.perf_furo_canto
    x0 = p.perf_x_frente - p.perf_l + p.perf_furo_canto
    y = p.perf_w / 2 - p.perf_furo_canto

            cil_z(p.tol.inserto_m3_d, p.tol.inserto_m3_prof, x, y, p.perf_z - p.tol.inserto_m3_prof)
```

```python
"""Base principal (spec §5.4). Impressa com a face de baixo na mesa."""
from __future__ import annotations

from functools import lru_cache

from build123d import Part, Plane, Rot, mirror

from params import Params, altura_eixo, carregar
from parts import battery_cradle, electronics, motor_clamp, sensor_bar
from parts.common import caixa_lim, cil_y, cil_z

ROT_IMPRESSAO = Rot(0, 0, 0)
FURO_CUBO = 11.4   # passagem do cubo ⌀10 da roda pela parede de batente (0,7 mm de folga)


def _x_tras(p: Params) -> float:
    return p.perf_x_frente - p.perf_l - 4.0


def _x_frente(p: Params) -> float:
    return sensor_bar.furos_base(p)[0][0] + 6.0


def _motor_lado(p: Params, lado: int) -> Part:
    zc = altura_eixo(p)
    w = p.n20_largura / 2 + p.tol.n20_folga
    ox = motor_clamp.orelha_x(p) + motor_clamp.ORELHA_L / 2
    yf = p.motor_face_y
    y0 = yf - p.n20_caixa_l
    topo_berco = zc - motor_clamp.APERTO
    # Aba da placa sob o motor até a parede de batente
    aba = caixa_lim(-ox, ox, p.base_meia_largura - 1, yf + p.parede_batente, p.base_z,
                    p.base_topo)
    # Berço da caixa de redução (metade de baixo) com furos piloto dos M2
    berco = caixa_lim(-ox, ox, y0, yf, p.base_topo, topo_berco) - \
        caixa_lim(-w, w, y0 - 1, yf + 1, zc - p.n20_altura / 2 - p.tol.n20_folga, topo_berco + 1)
    for x, y in motor_clamp.furos_parafuso(p, +1):
        berco -= cil_z(p.tol.autoatarraxante_m2, p.tol.autoatarraxante_prof, x, y,
                       topo_berco - p.tol.autoatarraxante_prof)
    # Apoio sob o corpo do motor (perto do encoder)
    y_apoio = yf - p.n20_total_l + 6.0
    apoio = caixa_lim(-w, w, y_apoio - 3, y_apoio + 3, p.base_topo,
                      zc - p.n20_altura / 2 - p.tol.n20_folga)
    # Parede de batente com passagem do cubo
    batente = caixa_lim(-w - p.parede, w + p.parede, yf, yf + p.parede_batente, p.base_topo,
                        zc + p.n20_altura / 2) - \
        cil_y(FURO_CUBO, p.parede_batente + 2, 0, yf - 1, zc)
    lado_esq = aba + berco + apoio + batente
    return lado_esq if lado > 0 else mirror(lado_esq, about=Plane.XZ)


@lru_cache(maxsize=None)
def fazer(p: Params | None = None) -> Part:
    p = p or carregar()
    placa = caixa_lim(_x_tras(p), _x_frente(p), -p.base_meia_largura, p.base_meia_largura,
                      p.base_z, p.base_topo)
    b = placa + _motor_lado(p, +1) + _motor_lado(p, -1)
    b += battery_cradle.feicao(p)
    b += electronics.colunas(p)
    b += electronics.suporte_chave(p)
    b -= electronics.rasgos_abracadeira(p)
    for x, y in sensor_bar.furos_base(p):
        b -= cil_z(p.tol.furo_m3, p.base_t + 2, x, y, p.base_z - 1)
    return b


def paredes(p: Params) -> dict[str, float]:
    return {"placa": p.base_t, "batente": p.parede_batente, "berco": p.parede,
            "coluna_ao_inserto": (electronics.COLUNA_D - p.tol.inserto_m3_d) / 2}
```

```python
"""Montagem do LFR v2 e verificações da spec §5.8."""
from __future__ import annotations

from itertools import combinations

from build123d import CenterOf, Part, Pos, Rot

from params import Params, altura_eixo
from parts import base, components, motor_clamp, sensor_bar, skid, wheel
from parts.common import caixa

TOL_VOLUME = 1e-3   # mm³ — abaixo disso é contato de face, não interferência


def pecas_impressas(p: Params, calcos: int = 0) -> dict[str, Part]:
    # base.fazer e wheel.fazer_roda usam lru_cache com Params congelado como chave
    pecas = {
        "base": base.fazer(p),
        "abraçadeira E": motor_clamp.fazer(p, +1),
        "abraçadeira D": motor_clamp.fazer(p, -1),
        "barra de sensores": sensor_bar.fazer(p, calcos),
        "patim E": skid.fazer(p, +1, calcos),
        "patim D": skid.fazer(p, -1, calcos),
    }
    if calcos:
        h = sensor_bar.barra_z0(p, 0)                 # topo do patim
        for lado, nome in ((+1, "calços patim E"), (-1, "calços patim D")):
            pecas[nome] = caixa(sensor_bar.PATIM_X, sensor_bar.PATIM_Y, 0.5 * calcos,
                                p.lookahead, lado * sensor_bar.y_patim(p), h)
        pecas["calços língua"] = Pos(0, 0, p.base_topo) * sensor_bar.calco_lingua(p, calcos)
    return pecas


def rodas(p: Params) -> dict[str, Part]:
    zc = altura_eixo(p)
    y = p.motor_face_y + p.folga_roda_motor + p.roda_cubo_l
    roda, oring = wheel.fazer_roda(p), wheel.fazer_oring(p)
    esq = Pos(0, y, zc) * Rot(90, 0, 0)     # +Z local (cubo) → -Y
    dir_ = Pos(0, -y, zc) * Rot(-90, 0, 0)  # +Z local (cubo) → +Y
    return {"roda E": esq * roda, "O-ring E": esq * oring,
            "roda D": dir_ * roda, "O-ring D": dir_ * oring}


def componentes(p: Params, calcos: int = 0) -> dict[str, Part]:
    return {
        "motor E": components.motor(p, +1),
        "motor D": components.motor(p, -1),
        "suporte 2×18650": components.suporte_bateria(p),
        "QTR-8A": components.qtr(p, calcos),
        "perfboard": components.perfboard(p),
    }


def _todas(p: Params, calcos: int) -> dict[str, Part]:
    return {**pecas_impressas(p, calcos), **rodas(p), **componentes(p, calcos)}


def interferencias(p: Params, calcos: int = 0) -> list[tuple[str, str, float]]:
    todas = _todas(p, calcos)
    ruins = []
    for (na, a), (nb, b) in combinations(todas.items(), 2):
        if {na[-1], nb[-1]} <= {"E", "D"} and na.split()[0] in ("roda", "O-ring") and \
                nb.split()[0] in ("roda", "O-ring") and na[-1] == nb[-1]:
            continue  # roda e seu próprio O-ring se tocam por construção
        v = (a & b).volume
        if v > TOL_VOLUME:
            ruins.append((na, nb, round(v, 3)))
    return ruins


def folga_minima_rodas(p: Params) -> float:
    giram = rodas(p)
    fixas = {**pecas_impressas(p), **componentes(p)}
    return min(r.distance_to(f) for r in giram.values() for f in fixas.values())


def massa_e_cg(p: Params):
    itens: list[tuple[str, float, tuple[float, float, float]]] = []
    for nome, peca in {**pecas_impressas(p), **rodas(p)}.items():
        if nome.startswith("O-ring"):
            continue
        m = peca.volume * p.petg_densidade * p.preenchimento
        c = peca.center(CenterOf.MASS)
        itens.append((nome, m, (c.X, c.Y, c.Z)))
    for nome, m, solido in components.massas(p):
        c = solido.center(CenterOf.MASS)
        itens.append((nome, m, (c.X, c.Y, c.Z)))
    total = sum(m for _, m, _ in itens)
    cg = tuple(sum(m * pos[i] for _, m, pos in itens) / total for i in range(3))
    return total, cg, [(n, round(m, 1)) for n, m, _ in itens]


def carga_patins(p: Params) -> float:
    """Fração do peso nos patins: momento em torno do eixo motriz (x = 0)."""
    _, (x_cg, _, _), _ = massa_e_cg(p)
    return x_cg / p.lookahead


def envelope(p: Params) -> tuple[float, float, float]:
    tudo = list(_todas(p, 0).values())
    xs = [b.bounding_box() for b in tudo]
    return (max(b.max.X for b in xs) - min(b.min.X for b in xs),
            max(b.max.Y for b in xs) - min(b.min.Y for b in xs),
            max(b.max.Z for b in xs) - min(b.min.Z for b in xs))
```

```python
from __future__ import annotations

import math

from build123d import Part, Rot

from params import Params
from parts import base, motor_clamp, sensor_bar, skid, wheel

PECAS = {

    "roda": (lambda p: wheel.fazer_roda(p), wheel.ROT_IMPRESSAO, wheel.paredes),
}

EXCECOES_PAREDE = {
    "calço": "peça de 0,5 mm por definição (ajuste fino da altura do sensor)",
    "calço língua": "idem, sob a língua da barra",
}

# Área máxima voltada para baixo (> 45°) acima da mesa, por peça, com a feição que justifica.
LIMITE_BALANCO = {
    "base": 100.0,       # abas do berço (4 × 6 × 1,5) + topo do furo ⌀11,4 nos batentes
                         # + teto da janela da chave (ponte de 8,8 mm)
    "abraçadeira": 20.0,  # topo dos furos M2 das orelhas (horizontais nesta orientação)
    "barra de sensores": 10.0,
    "patim": 5.0,
    "calço": 0.0,
    "calço língua": 0.0,
    "roda": 15.0,        # topo do furo radial M2
}


def na_orientacao(nome: str, p: Params) -> Part:
    fabrica, rot, _ = PECAS[nome]
    return na_mesa(fabrica(p), rot)


def area_balanco(part: Part, angulo: float = 45.0, tolerancia: float = 0.05) -> float:
    """Soma a área dos triângulos voltados para baixo além de `angulo` e acima da mesa."""
    vertices, triangulos = part.tessellate(tolerancia)
    limite = -math.cos(math.radians(angulo))
    z_mesa = part.bounding_box().min.Z + 0.2
    area = 0.0
    for i, j, k in triangulos:
        a, b, c = vertices[i], vertices[j], vertices[k]
        n = (b - a).cross(c - a)
        dupla = n.length
        if dupla == 0:
            continue
        if n.Z / dupla < limite and min(a.Z, b.Z, c.Z) > z_mesa:
            area += dupla / 2
    return area
```

```python
"""Cupom de tolerância (spec §5.7). Ler no manual de montagem como preencher o
[tolerancias] do medidas.toml a partir dele."""
from __future__ import annotations

from build123d import Align, Box, Cylinder, Part, Pos, Rot

from params import Params
from parts.common import MIN_Z, caixa, caixa_lim, cil_z

ROT_IMPRESSAO = Rot(0, 0, 0)
L, W, T = 70.0, 34.0, 3.0

FUROS = [2.8, 2.9, 3.0, 3.1, 3.2, 3.3, 3.4]
FUROS_D = [(3.00, 2.50), (3.05, 2.55), (3.10, 2.60)]
INSERTOS = [3.9, 4.0, 4.1]
BOLSOES = [0.2, 0.3, 0.4]
RASGOS = [1.8, 2.0, 2.2]

FEICOES = ([(f"furo {d:.1f}", d) for d in FUROS] +
           [(f"eixo D {d:.2f}/{a:.2f}", d) for d, a in FUROS_D] +
           [(f"inserto {d:.1f}", d) for d in INSERTOS] +
           [(f"bolsão +{f:.1f}", f) for f in BOLSOES] +
           [(f"rasgo PCB {w:.1f}", w) for w in RASGOS])


def _furo_d(d: float, achatado: float, x: float, y: float) -> Part:
    r = d / 2
    manter = Pos(x, y - r, -1) * Box(d + 1, achatado, T + 2,
                                     align=(Align.CENTER, Align.MIN, Align.MIN))
    return (Pos(x, y, -1) * Cylinder(r, T + 2, align=MIN_Z)) & manter


def _entalhes(n: int, x: float) -> Part:
    """n entalhes de 0,8 mm na borda de baixo (y = -W/2) para identificar a feição."""
    peca = None
    for i in range(n):
        e = caixa(0.6, 1.6, T + 2, x - (n - 1) * 0.6 + i * 1.2, -W / 2, -1)
        peca = e if peca is None else peca + e
    return peca


def fazer(p: Params) -> Part:
    c = caixa(L, W, T)
    x0 = -L / 2 + 5
    for i, d in enumerate(FUROS):
        x = x0 + i * 9.5
        c -= cil_z(d, T + 2, x, 9, -1)
        c -= _entalhes(i + 1, x)
    for i, (d, a) in enumerate(FUROS_D):
        c -= _furo_d(d, a, x0 + i * 9.5, 0)
    for i, d in enumerate(INSERTOS):
        c -= cil_z(d, T + 2, x0 + (i + 4) * 9.5, 0, -1)
    for i, f in enumerate(BOLSOES):
        s = 10.0 + 2 * f
        c -= caixa(s, s, T - 1.2, x0 + 4 + i * 14, -9, 1.2)
    for i, w in enumerate(RASGOS):
        c -= caixa_lim(L / 2 - 20 + i * 6 - w / 2, L / 2 - 20 + i * 6 + w / 2, -14, -4, 1.2,
                       T + 1)
    return c


def chave_teste() -> Part:
    """Quadrado de 10 × 10 × 2 mm, impresso junto, para testar os bolsões."""
    return caixa(10.0, 10.0, 2.0)
```

```python
#!/usr/bin/env python3
"""Gera hardware/cad/out/: STL (orientação de impressão), STEP (montagem) e relatório.

Uso: .venv/bin/python build.py
"""
from __future__ import annotations

import sys
from pathlib import Path

from build123d import Compound, export_step, export_stl

import assembly
import printability as pr
from params import CAD_DIR, Params, carregar
from parts import motor_clamp, tolerance_coupon, wheel
from parts.common import na_mesa

QUANTIDADE = {"base": 1, "abracadeira_E": 1, "abracadeira_D": 1, "barra_sensores": 1,
              "patim": 2, "calco": 4, "calco_lingua": 2, "roda": 2, "cupom": 1,
              "cupom_chave": 1}


def _pecas(p: Params) -> dict[str, tuple]:
    """nome → (peça na montagem, peça na mesa)."""
    imp = assembly.pecas_impressas(p)
    rodas = assembly.rodas(p)
    return {
        "base": (imp["base"], pr.na_orientacao("base", p)),
        "abracadeira_E": (imp["abraçadeira E"],
                          na_mesa(imp["abraçadeira E"], motor_clamp.ROT_IMPRESSAO)),
        "abracadeira_D": (imp["abraçadeira D"],
                          na_mesa(imp["abraçadeira D"], motor_clamp.ROT_IMPRESSAO)),
        "barra_sensores": (imp["barra de sensores"], pr.na_orientacao("barra de sensores", p)),
        "patim": (imp["patim E"], pr.na_orientacao("patim", p)),
        "calco": (pr.PECAS["calço"][0](p), pr.na_orientacao("calço", p)),
        "calco_lingua": (pr.PECAS["calço língua"][0](p), pr.na_orientacao("calço língua", p)),
        "roda": (rodas["roda E"], pr.na_orientacao("roda", p)),
        "cupom": (tolerance_coupon.fazer(p), na_mesa(tolerance_coupon.fazer(p),
                                                     tolerance_coupon.ROT_IMPRESSAO)),
        "cupom_chave": (tolerance_coupon.chave_teste(),
                        na_mesa(tolerance_coupon.chave_teste(), tolerance_coupon.ROT_IMPRESSAO)),
    }


def _relatorio(p: Params) -> str:
    total, (x, y, z), itens = assembly.massa_e_cg(p)
    g = wheel.verificar_oring(p)
    rpm = wheel.rpm_sob_carga(p)
    linhas = [
        "# Relatório de montagem — LFR v2",
        "",
        "Gerado por `hardware/cad/build.py`. Massas impressas estimadas (PETG "
        f"{p.petg_densidade * 1000:.2f} g/cm³, preenchimento {p.preenchimento:.0%}); "
        "massas compradas do `specs.py` (estimadas até a pesagem — D7).",
        "",
        "| Grandeza | Valor |",
        "|----------|-------|",
        f"| Massa total | {total:.0f} g |",
        f"| CG (x, y, z) | ({x:.1f}, {y:.1f}, {z:.1f}) mm — origem no eixo motriz, no chão |",
        f"| Carga nos patins | {assembly.carga_patins(p):.1%} do peso (meta 10–20 %) |",
        f"| Lookahead (sensores → eixo) | {p.lookahead:.0f} mm |",
        f"| Roda (diâmetro com O-ring) | {p.roda_od:.1f} mm |",
        f"| O-ring | {g.id:.0f} × {g.cs:.1f} mm, fundo do canal ⌀{g.fundo_d:.2f} mm |",
        f"| Velocidade estimada | {wheel.velocidade_m_s(p.roda_od, rpm):.2f} m/s "
        f"(N20 {p.n20_rpm_6v:.0f} RPM @ 6 V, vref {p.vref} V, carga {wheel.CARGA:.0%}) |",
        f"| Aceleração a meia velocidade | {wheel.aceleracao_m_s2(p, total):.1f} m/s² |",
        f"| Folga mínima das rodas | {assembly.folga_minima_rodas(p):.2f} mm |",
        "| Envelope | {:.0f} × {:.0f} × {:.0f} mm |".format(*assembly.envelope(p)),
        "",
        "## Massas",
        "",
        "| Item | g |",
        "|------|---|",
    ] + [f"| {n} | {m:.1f} |" for n, m in itens] + [
        "",
        "## Peças para imprimir",
        "",
        "| Arquivo | Quantidade |",
        "|---------|------------|",
    ] + [f"| `{n}.stl` | {q} |" for n, q in QUANTIDADE.items()]
    return "\n".join(linhas) + "\n"


def gerar(destino: Path, p: Params | None = None) -> dict[str, Path]:
    p = p or carregar()
    destino.mkdir(parents=True, exist_ok=True)
    saidas = {}
    for nome, (montagem, mesa) in _pecas(p).items():
        export_stl(mesa, str(destino / f"{nome}.stl"))
        export_step(montagem, str(destino / f"{nome}.step"))
        saidas[nome] = destino / f"{nome}.stl"
    tudo = {**assembly.pecas_impressas(p), **assembly.rodas(p), **assembly.componentes(p)}
    export_step(Compound(children=list(tudo.values())), str(destino / "assembly.step"))
    (destino / "relatorio_montagem.md").write_text(_relatorio(p), encoding="utf-8")
    return saidas


if __name__ == "__main__":
    gerar(CAD_DIR / "out")
    print((CAD_DIR / "out" / "relatorio_montagem.md").read_text())
    sys.exit(0)
```
